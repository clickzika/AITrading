Overview

AITrading uses TimescaleDB (PostgreSQL extension) for all persistent storage. TimescaleDB provides automatic partitioning of time-series data into chunks, significantly improving query performance on OHLCV candle history. Async access via SQLAlchemy 2.0 with asyncpg driver.

TableTypePurposeEst. Size
ohlcv_candlesHypertableOHLCV candle data per symbol/timeframe~50M rows/year
tradesRegularAll order intents + executed trades<10K rows/year
positionsRegularCurrent open positions (synced from MT5)<100 rows
risk_stateRegularDaily risk tracking (loss cap, halt flag)365 rows/year
signalsRegularAll strategy signal events (fired + rejected)<50K rows/year
smc_zonesRegularActive SMC zones per symbol/timeframe<1K rows

Table Schemas

ohlcv_candles — Hypertable

CREATE TABLE ohlcv_candles (
    time        TIMESTAMPTZ     NOT NULL,
    symbol      VARCHAR(20)     NOT NULL,
    timeframe   VARCHAR(5)      NOT NULL,   -- M1,M5,M15,M30,H1,H4,D1,W1
    open        NUMERIC(20, 8)  NOT NULL,
    high        NUMERIC(20, 8)  NOT NULL,
    low         NUMERIC(20, 8)  NOT NULL,
    close       NUMERIC(20, 8)  NOT NULL,
    volume      NUMERIC(20, 8)  NOT NULL DEFAULT 0,
    PRIMARY KEY (time, symbol, timeframe)
);

-- Convert to hypertable, partition by 1 day
SELECT create_hypertable(
    'ohlcv_candles',
    'time',
    chunk_time_interval => INTERVAL '1 day',
    if_not_exists => TRUE
);

trades

CREATE TABLE trades (
    id              UUID            PRIMARY KEY DEFAULT gen_random_uuid(),
    mt5_ticket      BIGINT          UNIQUE,           -- NULL until MT5 fills
    symbol          VARCHAR(20)     NOT NULL,
    strategy_id     VARCHAR(100)    NOT NULL,
    direction       VARCHAR(4)      NOT NULL CHECK (direction IN ('BUY','SELL')),
    lot_size        NUMERIC(10, 4)  NOT NULL,
    entry_price     NUMERIC(20, 8),                   -- NULL = market order
    sl_price        NUMERIC(20, 8)  NOT NULL,
    tp_price        NUMERIC(20, 8)  NOT NULL,
    exit_price      NUMERIC(20, 8),
    pnl_usd         NUMERIC(12, 2),
    status          VARCHAR(20)     NOT NULL DEFAULT 'PENDING'
                        CHECK (status IN ('PENDING','OPEN','CLOSED','CANCELLED','ERROR')),
    entry_reason    JSONB,                            -- indicator snapshot at entry
    error_message   TEXT,
    opened_at       TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
    closed_at       TIMESTAMPTZ,
    created_at      TIMESTAMPTZ     NOT NULL DEFAULT NOW()
);

positions

CREATE TABLE positions (

    lot_size        NUMERIC(10, 4)  NOT NULL,
    open_price      NUMERIC(20, 8)  NOT NULL,
    current_price   NUMERIC(20, 8),
    sl_price        NUMERIC(20, 8),
    tp_price        NUMERIC(20, 8),
    floating_pnl    NUMERIC(12, 2),
    opened_at       TIMESTAMPTZ     NOT NULL,
    synced_at       TIMESTAMPTZ     NOT NULL DEFAULT NOW()
);

risk_state

CREATE TABLE risk_state (
    date                DATE        PRIMARY KEY,
    realized_pnl_usd    NUMERIC(12, 2)  NOT NULL DEFAULT 0,
    unrealized_pnl_usd  NUMERIC(12, 2)  NOT NULL DEFAULT 0,
    trade_count         INTEGER         NOT NULL DEFAULT 0,
    trading_halted      BOOLEAN         NOT NULL DEFAULT FALSE,
    halt_reason         VARCHAR(100),
    peak_equity         NUMERIC(12, 2),             -- for drawdown calc

signals

CREATE TABLE signals (
    id              UUID            PRIMARY KEY DEFAULT gen_random_uuid(),
    strategy_id     VARCHAR(100)    NOT NULL,
    symbol          VARCHAR(20)     NOT NULL,
    timeframe       VARCHAR(5)      NOT NULL,
    direction       VARCHAR(4)      NOT NULL,
    status          VARCHAR(20)     NOT NULL  -- FIRED | REJECTED | EXECUTED
                        CHECK (status IN ('FIRED','REJECTED','EXECUTED')),
    reject_reason   VARCHAR(200),
    trade_id        UUID            REFERENCES trades(id),
    indicator_snap  JSONB,                    -- full IndicatorSnapshot at signal time
    fired_at        TIMESTAMPTZ     NOT NULL DEFAULT NOW()
);

smc_zones

CREATE TABLE smc_zones (
    id          UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
    symbol      VARCHAR(20) NOT NULL,
    timeframe   VARCHAR(5)  NOT NULL,
    zone_type   VARCHAR(20) NOT NULL  -- ORDER_BLOCK | FVG | DEMAND | SUPPLY | BOS | CHOCH
                    CHECK (zone_type IN ('ORDER_BLOCK','FVG','DEMAND','SUPPLY','BOS','CHOCH')),
    direction   VARCHAR(10) NOT NULL  -- BULLISH | BEARISH
                    CHECK (direction IN ('BULLISH','BEARISH')),
    price_high  NUMERIC(20, 8) NOT NULL,
    price_low   NUMERIC(20, 8) NOT NULL,
    is_valid    BOOLEAN     NOT NULL DEFAULT TRUE,
    formed_at   TIMESTAMPTZ NOT NULL,
    invalidated_at  TIMESTAMPTZ,
    created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- smc_zones: active zone lookup
CREATE INDEX idx_smc_symbol_tf_valid ON smc_zones (symbol, timeframe, is_valid)
    WHERE is_valid = TRUE;

SQLAlchemy Models

from __future__ import annotations
from datetime import datetime
from decimal import Decimal
from uuid import UUID, uuid4
import enum
from sqlalchemy import String, Numeric, Boolean, Text, ForeignKey
from sqlalchemy import TIMESTAMP, Integer, BigInteger, Date, CheckConstraint
from sqlalchemy.dialects.postgresql import UUID as PGUUID, JSONB
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, relationship

class Base(DeclarativeBase):
    pass

class TradeStatus(str, enum.Enum):
    PENDING   = "PENDING"
    OPEN      = "OPEN"
    CLOSED    = "CLOSED"
    CANCELLED = "CANCELLED"
    ERROR     = "ERROR"

class OHLCVCandle(Base):
    __tablename__ = "ohlcv_candles"
    time:       Mapped[datetime] = mapped_column(TIMESTAMP(timezone=True), primary_key=True)
    symbol:     Mapped[str]      = mapped_column(String(20), primary_key=True)
    timeframe:  Mapped[str]      = mapped_column(String(5),  primary_key=True)
    open:       Mapped[Decimal]  = mapped_column(Numeric(20, 8))
    high:       Mapped[Decimal]  = mapped_column(Numeric(20, 8))
    low:        Mapped[Decimal]  = mapped_column(Numeric(20, 8))
    close:      Mapped[Decimal]  = mapped_column(Numeric(20, 8))
    volume:     Mapped[Decimal]  = mapped_column(Numeric(20, 8), default=0)

class Trade(Base):
    __tablename__ = "trades"
    id:             Mapped[UUID]         = mapped_column(PGUUID, primary_key=True, default=uuid4)
    mt5_ticket:     Mapped[int | None]   = mapped_column(BigInteger, unique=True)
    symbol:         Mapped[str]          = mapped_column(String(20), nullable=False)
    strategy_id:    Mapped[str]          = mapped_column(String(100), nullable=False)
    direction:      Mapped[str]          = mapped_column(String(4), nullable=False)
    lot_size:       Mapped[Decimal]      = mapped_column(Numeric(10, 4), nullable=False)
    entry_price:    Mapped[Decimal|None] = mapped_column(Numeric(20, 8))
    sl_price:       Mapped[Decimal]      = mapped_column(Numeric(20, 8), nullable=False)
    tp_price:       Mapped[Decimal]      = mapped_column(Numeric(20, 8), nullable=False)
    exit_price:     Mapped[Decimal|None] = mapped_column(Numeric(20, 8))
    pnl_usd:        Mapped[Decimal|None] = mapped_column(Numeric(12, 2))
    status:         Mapped[str]          = mapped_column(String(20), default="PENDING")
    entry_reason:   Mapped[dict|None]    = mapped_column(JSONB)
    error_message:  Mapped[str|None]     = mapped_column(Text)
    opened_at:      Mapped[datetime]     = mapped_column(TIMESTAMP(timezone=True))
    closed_at:      Mapped[datetime|None]= mapped_column(TIMESTAMP(timezone=True))
    created_at:     Mapped[datetime]     = mapped_column(TIMESTAMP(timezone=True))

class RiskState(Base):
    __tablename__ = "risk_state"
    date:               Mapped[datetime]     = mapped_column(Date, primary_key=True)
    realized_pnl_usd:   Mapped[Decimal]      = mapped_column(Numeric(12, 2), default=0)
    trading_halted:     Mapped[bool]         = mapped_column(Boolean, default=False)
    halt_reason:        Mapped[str|None]     = mapped_column(String(100))
    peak_equity:        Mapped[Decimal|None] = mapped_column(Numeric(12, 2))
    updated_at:         Mapped[datetime]     = mapped_column(TIMESTAMP(timezone=True))

env.py — Async Config

from alembic import context
from sqlalchemy.ext.asyncio import create_async_engine
from app.models import Base
from app.core.config import settings

def run_migrations_online():
    connectable = create_async_engine(settings.DATABASE_URL)
    with connectable.connect() as connection:
        context.configure(connection=connection, target_metadata=Base.metadata)
        with context.begin_transaction():
            context.run_migrations()

Query Patterns

Fetch Last N Candles (optimized)

SELECT time, open, high, low, close, volume
FROM ohlcv_candles
WHERE symbol = 'XAUUSD' AND timeframe = 'H1'
ORDER BY time DESC
LIMIT 500;
-- TimescaleDB uses chunk exclusion — only scans relevant partitions

Candle Range Query

SELECT * FROM ohlcv_candles
WHERE symbol = $1 AND timeframe = $2
  AND time >= $3 AND time < $4
ORDER BY time ASC;

Daily P&L Aggregation

SELECT DATE(closed_at) as day,
       COUNT(*) as trade_count,
       SUM(pnl_usd) as daily_pnl,
       COUNT(*) FILTER (WHERE pnl_usd > 0) as wins
FROM trades
WHERE status = 'CLOSED' AND closed_at >= NOW() - INTERVAL '30 days'
GROUP BY 1 ORDER BY 1;

Equity Curve

SELECT closed_at, SUM(pnl_usd) OVER (ORDER BY closed_at) as equity
FROM trades
WHERE status = 'CLOSED'
ORDER BY closed_at;
