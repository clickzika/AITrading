# backend/tests/unit/test_indicators.py
import pandas as pd
import pytest
from app.services.analysis.indicators import compute_ema, compute_macd, detect_ema_cross

@pytest.fixture
def sample_ohlcv():
    """200 candles of synthetic price data."""
    import numpy as np
    np.random.seed(42)
    prices = 1800 + np.cumsum(np.random.randn(200) * 0.5)
    return pd.DataFrame({"close": prices})

def test_ema_length(sample_ohlcv):
    result = compute_ema(sample_ohlcv["close"], period=20)
    assert len(result) == len(sample_ohlcv)
    assert result.notna().sum() == len(sample_ohlcv) - 19  # first 19 are NaN

def test_ema_cross_bullish(sample_ohlcv):
    # Force a cross at index 50
    sample_ohlcv.loc[49, "close"] = 1700  # make fast below slow
    sample_ohlcv.loc[50, "close"] = 1900  # force cross up
    snap = detect_ema_cross(sample_ohlcv["close"], fast=5, slow=20)
    assert snap in ("bullish_cross", "bullish", "bearish", "bearish_cross", None)

def test_macd_histogram_sign():
    """MACD histogram should be positive when MACD > signal."""
    data = pd.Series([100 + i * 0.5 for i in range(100)])
    result = compute_macd(data)
    last = result.iloc[-1]
    if last["macd_line"] > last["macd_signal"]:
        assert last["macd_hist"] > 0

MT5 Mock Strategy

MetaTrader5 Python lib is Windows-only and requires a running terminal. Mock it completely in tests using unittest.mock — never call the real lib in unit/integration tests.

# tests/mocks/mt5_mock.py
from unittest.mock import MagicMock, AsyncMock
from dataclasses import dataclass

@dataclass
class MockAccountInfo:
    balance: float = 10000.0
    equity: float = 10000.0
    margin: float = 0.0
    free_margin: float = 10000.0
    currency: str = "USD"

@dataclass
class MockOrderResult:
    retcode: int = 10009   # TRADE_RETCODE_DONE
    order: int = 12345678  # mt5 ticket
    volume: float = 0.01
    price: float = 2350.0

def make_mt5_client_mock():
    mock = AsyncMock()
    mock.connect.return_value = True
    mock.get_account_info.return_value = MockAccountInfo()
    mock.place_market_order.return_value = MockOrderResult()
    mock.get_open_positions.return_value = []
    return mock
# Usage in test
from tests.mocks.mt5_mock import make_mt5_client_mock

async def test_order_manager_writes_to_db_before_mt5(db_session):
    mt5_mock = make_mt5_client_mock()
    order_mgr = OrderManager(mt5_client=mt5_mock, db=db_session)
    intent = OrderIntent(symbol="XAUUSD", direction="BUY", lot_size=0.01,
                         sl_price=2320.0, tp_price=2410.0, strategy_id="test")
    await order_mgr.execute(intent)
    # Verify DB write happened BEFORE MT5 call
    trade = await db_session.get(Trade, intent.id)
    assert trade is not None
    assert trade.status == "OPEN"
    mt5_mock.place_market_order.assert_awaited_once()

Webhook Tests

# test_webhook.py
import hmac, hashlib

def test_webhook_accepts_valid_signature(client, monkeypatch):
    secret = "test-secret"
    monkeypatch.setenv("TRADINGVIEW_WEBHOOK_SECRET", secret)
    payload = b'{"symbol":"XAUUSD","action":"BUY","price":2350,"strategy_id":"tv_test"}'
    sig = "sha256=" + hmac.new(secret.encode(), payload, hashlib.sha256).hexdigest()
    resp = client.post("/api/v1/webhook/tradingview",
                       content=payload,
                       headers={"X-Signature": sig, "Content-Type": "application/json"})
    assert resp.status_code == 200

def test_webhook_rejects_bad_signature(client):
    resp = client.post("/api/v1/webhook/tradingview",
                       json={"symbol": "XAUUSD", "action": "BUY"},
                       headers={"X-Signature": "sha256=badhash"})
    assert resp.status_code == 401

CI Pipeline (GitHub Actions)

# .github/workflows/ci.yml
name: CI
on: [push, pull_request]

jobs:
  backend-tests:
    runs-on: ubuntu-latest   # Note: MT5 tests skipped (Windows-only)
    services:
      postgres:
        image: timescale/timescaledb:latest-pg15
        env:
          POSTGRES_DB: aitrading_test
          POSTGRES_USER: aitrading
          POSTGRES_PASSWORD: test
        ports: ["5432:5432"]
      redis:
        image: redis:7-alpine
        ports: ["6379:6379"]
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: {python-version: '3.11'}
      - run: pip install -r backend/requirements-test.txt
      - run: cd backend && pytest --cov=app tests/ -v
        env:
          DATABASE_URL: postgresql+asyncpg://aitrading:test@localhost:5432/aitrading_test
          REDIS_URL: redis://localhost:6379/0
          API_SECRET_KEY: test-secret-key-for-ci
          TRADING_MODE: demo
          MT5_MOCK: "true"   # flag to skip MT5 lib import

  frontend-build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: {node-version: '18'}
      - run: cd frontend && npm ci && npm run build && npm run type-check
